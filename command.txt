cd "C:\Users\Pond\Documents\GitHub\SIP-record-video"

$root = "$PWD\third_party\openh264"
$build = "$root\build_x64"
if (!(Test-Path $build)) {
  cmake -S $root -B $build -G "Visual Studio 17 2022" -A x64 -DBUILD_SHARED_LIBS=ON -DCMAKE_BUILD_TYPE=Release
  cmake --build $build --config Release --parallel
}
$dll = Get-ChildItem -Recurse $build -Filter 'openh264*.dll' -ErrorAction SilentlyContinue | Where-Object { $_.FullName -match '\\Release\\' } | Select-Object -First 1
$lib = Get-ChildItem -Recurse $build -Filter 'openh264*.lib' -ErrorAction SilentlyContinue | Where-Object { $_.FullName -match '\\Release\\' } | Select-Object -First 1
if (!$dll) { throw "OpenH264 DLL still not found (searching $build)" }
if (!$lib) { throw "OpenH264 LIB still not found (searching $build)" }
Copy-Item $dll.FullName "$build\openh264-8.dll" -Force
Copy-Item $lib.FullName "$build\openh264.lib" -Force

powershell -ExecutionPolicy Bypass -File .\scripts\setup-windows.ps1

Commands:
  d <ext|sip-uri>  : dial (video enabled)
  a                : answer incoming (200 OK, with video)
  h                : hangup active call
  w <path.wav>     : play WAV into call
  tx <gain>        : set TX level (e.g. 1.10)
  rx <gain>        : set RX level (e.g. 1.05)
  pv [cap_id]      : start local camera preview (default cap 0)
  pvoff [cap_id]   : stop local camera preview (default cap 0)
  lsvid            : list video devices (cap/render)
  q                : quit
